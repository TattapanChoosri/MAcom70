เปิด Google Sheet
วิธีเปิดใช้งานรับข้อมูลใน 1 นาที
  1. เปิดไฟล์ Google Sheet ด้านบน แล้วไปที่เมนู ส่วนขยาย (Extensions) → Apps Script
  2. วางโค้ดด้านล่างแทนที่โค้ดเดิมทั้งหมด แล้วกดบันทึก
  3. กด ทำให้ใช้งานได้ (Deploy) → รายการใหม่ (New deployment) เลือกประเภท เว็บแอป (Web app) ตั้งผู้มีสิทธิ์เป็น ทุกคน (Anyone)
  4. คัดลอก Web app URL มาวางในช่องด้านบนแล้วกด บันทึกลิงก์
โค้ด Google Apps Script (Code.gs · 19 คอลัมน์) 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('ข้อมูลการเดินทาง');
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.timestamp || new Date().toLocaleString('th-TH'),
    data.title || '',
    data.firstNameEn || '',
    data.middleNameEn || '',
    data.lastNameEn || '',
    data.fullNameTh || '',
    data.dob || '',
    "'" + (data.phone || ''),
    data.email || '',
    "'" + (data.ropNumber || ''),
    data.specialMeal || '',
    data.seatAssist || '',
    data.outboundAirport || '',
    data.outboundAirline || '',
    data.outboundTime || '',
    data.returnAirport || '',
    data.returnAirline || '',
    data.returnTime || '',
    data.notes || ''
  ]);
  return ContentService
    .createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
