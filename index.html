โค้ด Google Apps Script (Code.gs · 18 คอลัมน์) 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('ข้อมูลการเดินทาง');
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([
    data.timestamp || new Date().toLocaleString('th-TH'),
    data.title || '',
    data.firstNameEn || '',
    data.middleNameEn || '',
    data.lastNameEn || '',
    data.fullNameTh || '',
    data.dob || '',
    "'" + (data.phone || ''),
    data.email || '',
    "'" + (data.ropNumber || ''),
    data.seatAssist || '',
    data.outboundAirport || '',
    data.outboundAirline || '',
    data.outboundTime || '',
    data.returnAirport || '',
    data.returnAirline || '',
    data.returnTime || '',
    data.notes || ''
  ]);
  return ContentService
    .createTextOutput(JSON.stringify({ status: 'success' }))
    .setMimeType(ContentService.MimeType.JSON);
}
